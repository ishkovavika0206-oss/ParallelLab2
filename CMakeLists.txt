cmake_minimum_required(VERSION 3.16)
project(ParallelLab2GCD LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(ParallelLab2GCD main.cpp)
target_compile_features(ParallelLab2GCD PRIVATE cxx_std_17)
target_link_libraries(ParallelLab2GCD PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(ParallelLab2GCD PRIVATE /utf-8 /W4)
else()
    target_compile_options(ParallelLab2GCD PRIVATE -Wall -Wextra -Wpedantic)
endif()
configure_file(input.txt input.txt COPYONLY)
set_property(TARGET ParallelLab2GCD PROPERTY VS_DEBUGGER_WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
